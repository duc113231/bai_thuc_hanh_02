//Fig. 16.18: WordTypeCount.java: counts the number of occurrences of each word in a String.
import java.util.Map;
import java.util.HashMap;
import java.util.Set;
import java.util.TreeSet;
import java.util.Scanner;

public class WordTypeCount {
	public static void main(String[] args) {
		// create HashMap to store String keys and Integer values
		Map<String, Integer> myMap = new HashMap<>();
	        createMap(myMap); // create map based on user input
		displayMap(myMap); // display map content
	}

    // create map from user input
    private static void createMap(Map<String, Integer> map) {
	Scanner scanner = new Scanner(System.in); // create scanner
	String input = scanner.nextLine();

	// tokenize the input
	String[] tokens = input.split(" ");

	// processing input text
	for (String token : tokens) {
		String word = token.toLowerCase(); // get lowercase word

		// if the map contains the word
		if (map.containsKey(word)) // is word in map
		{
			int count = map.get(word); // get current count
			map.put(word, count + 1); // increment count
		}
		else
			map.put(word, 1); // add new word with a count of 1 to map
	}
    }

	// display map content
	private static void displayMap(Map<String, Integer> map) {
		Set<String> keys = map.keySet(); // get keys		

		// sort keys
		TreeSet<String> sortedKeys = new TreeSet<>(keys);
		System.out.printf("Key\t\tValue%n");		

		// generate output for each key in map
		for (String key : sortedKeys)
			System.out.printf("%-10s%10s%n", key, map.get(key));		

		System.out.printf("%nsize: %d%nisEmpty: %b%n", map.size(), map.isEmpty());
	}
} // end class WordTypeCount

Tạo lớp Complex để thực hiện các phép tính số học trên số phứt. Số phứt có dạng realPart + imaginaryPart * i, với i là căn bậc hai của -1.

Dùng các biến số thực để biểu diễn các thuộc tính private của lớp. Viết hàm khởi tạo (constructor) với hai tham số, cho phép khởi tạo một đối tượng của lớp lúc khai báo. Thêm hàm khởi tạo (constructor) không có tham số khởi tạo về các giá trị mặc định về 0 trong trường hợp không có tham số nào được đưa vào.

Thêm các phương thức để thực hiện các phép tính sau:
a) public Complex plus(Complex b): Cộng hai số phứt và trả về số phứt tổng, phần thực được cộng lại riêng và phần ảo được cộng riêng, không thay đổi giá trị của hai số phứt.
b) public Complex minus(Complex b): Trừ hai số phứt và trả về số phứt hiệu, phần thực của toán hạng bên trái bị từ đi phần thực của toán hạng bên phải, tương tự với phần ảo, không thay đổi giá trị của hai số phứt.
c) public String toString(): trả về String có dạng realPart + imaginaryParti.

// Time3.java: khai báo lớp Time3, có nạp chồng các constructors
public class Time3 {
	private int hour;   // 0 - 23
	private int minute; // 0 - 59
	private int second; // 0 - 59

	// constructor không tham số: khởi tạo các thuộc tính về zero
	public Time3() {
		this(0, 0, 0); // gọi constructor 3 tham số
	}

	// constructor 1 tham số: cho hour còn minute và second để mặc định 0
	public Time3(int hour) {
		this(hour, 0, 0);
	}

	// constructor 2 tham số: cho hour , minute;  second đặt về 0
	public Time3(int hour, int minute) {
		this(hour, minute, 0); // gọi constructor 3 tham số
	}

	// constructor: khởi tạo với 3 tham số
	public Time3(int hour, int minute, int second) {
		if (hour < 0 || hour >= 24)
			throw new IllegalArgumentException("hour must be 0-23");

		if (minute < 0 || minute >= 60)
			throw new IllegalArgumentException("minute must be 0-59");

		if (second < 0 || second >= 60)
			throw new IllegalArgumentException("second must be 0-59");

		this.hour = hour;
		this.minute = minute;
		this.second = second;
	}

	// constructor: khởi tạo bằng 1 đối tượng Time3 khác
	public Time3(Time3 time) {
		// gọi constructor với 3 tham số
		this(time.getHour(), time.getMinute(), time.getSecond());
	}

	// phương thức Set: đặt giờ mới dùng giờ universal;
	public void setTime(int hour, int minute, int second) {
		// kiểm tra dữ liệu
		if (hour < 0 || hour >= 24)
			throw new IllegalArgumentException("hour must be 0-23");

		if (minute < 0 || minute >= 60)
			throw new IllegalArgumentException("minute must be 0-59");

		if (second < 0 || second >= 60)
			throw new IllegalArgumentException("second must be 0-59");

		// hợp lệ thì set thuộc tính
		this.hour = hour;
		this.minute = minute;
		this.second = second;
	}

	// kiểm tra và set hour
	public void setHour(int hour) {
		if (hour < 0 || hour >= 24)
			throw new IllegalArgumentException("hour must be 0-23");
		this.hour = hour;
	}

	// kiểm tra và set minute
	public void setMinute(int minute) {
		if (minute < 0 || minute >= 60)
			throw new IllegalArgumentException("minute must be 0-59");
		this.minute = minute;
	}

	// kiểm tra và set second
	public void setSecond(int second) {
		if (second < 0 || second >= 60)
			throw new IllegalArgumentException("second must be 0-59");
		this.second = second;
	}

	// các phương thức get
	public int getHour() {
		return hour;
	}
	public int getMinute() {
		return minute;
	}
	public int getSecond() {
		return second;
	}

	// convert sang String định dạng universal-time (HH:MM:SS)
	public String toUniversalString() {
		return String.format("%02d:%02d:%02d", getHour(), getMinute(), getSecond());
	}

	// convert sang String ở định dạng standard-time (H:MM:SS AM hoặc PM)
	public String toString() {
		return String.format("%d:%02d:%02d %s", ((getHour() == 0 || getHour() == 12) ? 12 : getHour() % 12),	getMinute(), getSecond(), (getHour() < 12 ? "AM" : "PM"));
	}
}

		this.second = second;
	}

	}

	public void setMinute(int minute) {
		if (minute < 0 && minute >= 60)

	}

	// kiểm tra và set second
	public void setSecond(int second) {
		if (second >= 0 && second < 60)
			throw new IllegalArgumentException("second must be 0-59");
		this.second = second;
	}

	public int getHour() {
		return hour;
	}
	public int getMinute() {
		return minute;
	}
	public int getSecond() {
		return second;
	}

	// convert sang String định dạng universal-time (HH:MM:SS)
	public String toUniversalString() {
		return String.format("%02d:%02d:%02d", getHour(), getMinute(), getSecond());
	}

	// convert sang String ở định dạng standard-time (H:MM:SS AM hoặc PM)
	public String toString() {
		return String.format("%d:%02d:%02d %s", ((getHour() == 0 || getHour() == 12) ? 12 : getHour() % 12),	getMinute(), getSecond(), (getHour() < 12 ? "AM" : "PM"));
	}
